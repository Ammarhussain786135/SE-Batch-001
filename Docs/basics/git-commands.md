# BurnDown

- git status (show the current state of the working directory)
- git init (initialize a new Git repository)
- git clone {repository-url} (clone an existing repository to your local machine)
- git remote -v (show the remote repository URLs)
- git remote add origin {repository-url} (add a remote repository named origin)
- git remote remove origin (remove the origin remote repository)
- git branch (show local branches)
- git branch {branch-name} (create a new branch)
- git branch -a (show all local and remote branches)
- git checkout {branch-name} (switch to an existing branch)
- git checkout -b {new-branch-name} (create and switch to a new branch)
- git switch {branch-name} (switch to an existing branch)
- git switch -c {new-branch-name} (create and switch to a new branch)
- git add {selected-files} (stage selected files for commit)
- git add . (stage all changed files in the current directory)
- git add -A (stage all changes including new, modified, and deleted files)
- git restore {file-name} (discard unstaged changes from a specific file)
- git restore . (discard all unstaged changes)
- git diff (show unstaged changes)
- git diff --staged (show changes that are staged for commit)
- git commit -m "{commit-message}" (create a commit with a message)
- git commit --amend (modify the most recent commit)
- git log (show the commit history)
- git log --oneline (show a compact one-line commit history)
- git show {commit-id} (show details and changes of a specific commit)
- git push (push local commits to the remote repository)
- git push origin {branch-name} (push a specific branch to origin)
- git push -u origin {branch-name} (push a branch and set its upstream remote)
- git pull (fetch and merge the latest changes from the remote repository)
- git fetch (download the latest remote changes without merging them)
- git merge {branch-name} (merge another branch into the current branch)
- git rebase {branch-name} (reapply current branch commits on top of another branch)
- git rebase --continue (continue a rebase after resolving conflicts)
- git rebase --abort (cancel the current rebase operation)
- git stash (temporarily save uncommitted changes)
- git stash push -m "{stash-message}" (stash changes with a custom message)
- git stash list (show all saved stashes)
- git stash pop (apply the latest stash and remove it from the stash list)
- git stash apply (apply a stash without removing it from the stash list)
- git stash drop (delete a specific stash)
- git cherry-pick {commit-id} (apply a specific commit to the current branch)
- git reset {commit-id} (move HEAD to a specific commit and reset staging)
- git reset --soft {commit-id} (move HEAD while keeping changes staged)
- git reset --mixed {commit-id} (move HEAD while keeping changes unstaged)
- git reset --hard {commit-id} (move HEAD and discard tracked changes)
- git revert {commit-id} (create a new commit that reverses a specific commit)
- git rm {file-name} (remove a file and stage the deletion)
- git mv {old-file-name} {new-file-name} (move or rename a file and stage the change)
- git tag (show all Git tags)
- git tag {tag-name} (create a new tag)
- git push origin {tag-name} (push a specific tag to the remote repository)
- git config --global user.name "{your-name}" (set your global Git username)
- git config --global user.email "{your-email}" (set your global Git email)
- git config --list (show Git configuration settings)
- git remote set-url origin {new-repository-url} (change the URL of the origin remote)
- git branch -d {branch-name} (delete a local branch safely)
- git branch -D {branch-name} (force delete a local branch)
- git push origin --delete {branch-name} (delete a remote branch)
- git clean -n (preview untracked files that would be removed)
- git clean -f (remove untracked files)
- git reflog (show the history of HEAD and branch reference changes)
- git help {command} (show help and documentation for a Git command)
