# BurnDown

- pwd (show the current working directory)
- ls (list files and directories)
- ls -l (show files with detailed information)
- ls -a (show all files including hidden files)
- ls -la (show all files with detailed information)
- cd {directory-name} (change to a specific directory)
- cd .. (move to the parent directory)
- cd ~ (move to the home directory)
- cd / (move to the root directory)
- mkdir {directory-name} (create a new directory)
- mkdir -p {parent-directory}/{directory-name} (create nested directories)
- rmdir {directory-name} (remove an empty directory)
- touch {file-name} (create a new empty file)
- cp {source-file} {destination} (copy a file)
- cp -r {source-directory} {destination} (copy a directory and its contents)
- mv {source} {destination} (move or rename a file or directory)
- rm {file-name} (remove a file)
- rm -r {directory-name} (remove a directory and its contents)
- rm -rf {directory-name} (forcefully remove a directory and its contents)
- cat {file-name} (display the contents of a file)
- less {file-name} (view a file one page at a time)
- head {file-name} (show the first lines of a file)
- head -n {number} {file-name} (show a specific number of lines from the beginning)
- tail {file-name} (show the last lines of a file)
- tail -f {file-name} (continuously monitor new content added to a file)
- nano {file-name} (open a file in the Nano text editor)
- vim {file-name} (open a file in the Vim text editor)
- echo "{text}" (print text in the terminal)
- echo "{text}" > {file-name} (write text to a file and overwrite existing content)
- echo "{text}" >> {file-name} (append text to a file)
- clear (clear the terminal screen)
- history (show previously executed commands)
- man {command} (show the manual page for a command)
- which {command} (show the location of an installed command)
- whoami (show the currently logged-in username)
- hostname (show the system hostname)
- uname -a (show detailed system information)
- date (show the current date and time)
- df -h (show disk space usage in a human-readable format)
- du -sh {directory-name} (show the total size of a directory)
- free -h (show available and used memory)
- top (show running processes and system resource usage)
- ps (show processes running in the current terminal)
- ps aux (show all running processes)
- kill {process-id} (terminate a process using its process ID)
- kill -9 {process-id} (forcefully terminate a process)
- grep "{text}" {file-name} (search for text inside a file)
- grep -r "{text}" {directory} (recursively search for text in a directory)
- find {directory} -name "{file-name}" (find files or directories by name)
- locate {file-name} (quickly search for a file by name)
- sort {file-name} (sort the lines of a file)
- uniq {file-name} (remove consecutive duplicate lines)
- wc {file-name} (count lines, words, and characters)
- wc -l {file-name} (count the number of lines in a file)
- chmod {permissions} {file-name} (change file or directory permissions)
- chmod +x {file-name} (make a file executable)
- chown {user}:{group} {file-name} (change file ownership and group)
- sudo {command} (run a command with administrator privileges)
- apt update (update package information on Debian or Ubuntu)
- apt upgrade (upgrade installed packages on Debian or Ubuntu)
- apt install {package-name} (install a package)
- apt remove {package-name} (remove an installed package)
- apt search {package-name} (search for available packages)
- systemctl status {service-name} (check the status of a system service)
- systemctl start {service-name} (start a system service)
- systemctl stop {service-name} (stop a system service)
- systemctl restart {service-name} (restart a system service)
- ping {hostname-or-ip} (test network connectivity)
- curl {url} (make a request to a URL)
- wget {url} (download a file from a URL)
- ip addr (show network interface and IP address information)
- ss -tuln (show listening network ports)
- ssh {user}@{host} (connect to a remote machine using SSH)
- scp {file} {user}@{host}:{destination} (copy a file to a remote machine using SSH)
- tar -czvf {archive-name}.tar.gz {directory} (create a compressed tar archive)
- tar -xzvf {archive-name}.tar.gz (extract a compressed tar archive)
- zip {archive-name}.zip {file-name} (create a ZIP archive)
- unzip {archive-name}.zip (extract a ZIP archive)
- env (show environment variables)
- printenv {variable-name} (show a specific environment variable)
- export {VARIABLE}={value} (set an environment variable for the current shell session)
- alias {name}="{command}" (create a shortcut for a command)
- history -c (clear the current shell history)
- exit (close the current terminal session)
- reboot (restart the system)
- shutdown now (shut down the system immediately)
